        function displayRestaurants(list) {
            const grid = document.getElementById('restaurantGrid');
            grid.innerHTML = '';

            if (list.length === 0) {
                grid.innerHTML = `<p style="grid-column: 1/-1; text-align: center; color: #888; padding: 2rem;">검색 결과가 없습니다 🥲</p>`;
                return;
            }

            list.forEach(resto => {
                const card = document.createElement('div');
                card.className = 'card';
                card.onclick = () => openModal(resto);
                card.innerHTML = `
                    <img src="${resto.image}" alt="${resto.name}">
                    <div class="card-content">
                        <div class="card-title">
                            <span>${resto.name}</span>
                            <span class="card-category">${resto.category}</span>
                        </div>
                        <p class="card-desc">${resto.desc}</p>
                        <div class="card-footer">
                            <span>📍 ${resto.location.split(',')[0]}</span>
                            <span class="rating">⭐ ${resto.rating}</span>
                        </div>
                    </div>
                `;
                grid.appendChild(card);
            });
        }

        function openModal(resto) {
            document.getElementById('modalTitle').innerText = resto.name;
            
            let menuHtml = '';
            resto.menus.forEach(menu => {
                menuHtml += `<li class="menu-item"><span>${menu.name}</span><strong>${menu.price}</strong></li>`;
            });

            const bodyHtml = `
                <p style="color: #666; margin-bottom: 1rem; font-size: 0.95rem;">${resto.desc}</p>
                <p style="margin-bottom: 1rem; font-size: 0.9rem;"><strong>주소:</strong> ${resto.location}</p>
                <div class="modal-section-title">📜 주요 메뉴 및 가격</div>
                <ul class="menu-list">${menuHtml}</ul>
                <div class="modal-section-title">💡 방문자 리뷰 및 특징 요약</div>
                <div class="review-box">${resto.review}</div>
            `;

            document.getElementById('modalBody').innerHTML = bodyHtml;
            document.getElementById('detailModal').style.display = 'flex';
        }

        function closeModal() {
            document.getElementById('detailModal').style.display = 'none';
        }

        window.onclick = function(event) {
            const modal = document.getElementById('detailModal');
            if (event.target === modal) {
                closeModal();
            }
        }

        function searchRestaurant() {
            const keyword = document.getElementById('searchInput').value.toLowerCase();
            const filtered = restaurants.filter(resto => 
                resto.name.toLowerCase().includes(keyword) || 
                resto.category.toLowerCase().includes(keyword) ||
                resto.desc.toLowerCase().includes(keyword) ||
                resto.location.toLowerCase().includes(keyword)
            );
            displayRestaurants(filtered);
        }

        displayRestaurants(restaurants);
